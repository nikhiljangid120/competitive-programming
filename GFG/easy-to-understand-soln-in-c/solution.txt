vector<int>dp;
    int smart_thief(vector<int>&arr,int idx)
    {
        //bse cse
        if(idx>=arr.size())
        {
            return 0;
        }
            
        if(dp[idx]!= -1) return dp[idx];
        // two choice
        int take = arr[idx]+ smart_thief(arr, idx+2);// idx+2 alternative ghr chori noi kr skte
                                                        // alternative (ek kai bdd kai)
        int skip= smart_thief(arr, idx+1);
        
        return dp[idx]=max(take,skip);
    }
    int findMaxSum(vector<int>& arr) {
        // code here
        dp.assign(arr.size(),-1);
        return smart_thief(arr,0);
    }
